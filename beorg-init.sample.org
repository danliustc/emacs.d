#+TITLE: beorg 配置
#+STARTUP: overview

* TODO 关键词

和桌面端保持一致：日常只用 TODO / DONE，WAITING、SOMEDAY 需要时才用。
刻意没有 NEXT。

#+begin_src scheme
(set! org-todo-action-keywords '("TODO" "WAITING" "SOMEDAY"))

(set! org-todo-done-keywords
      '("DONE" "CANCELLED"))
#+end_src

* Agenda 设置

#+begin_src scheme
;; init.org 是配置、archive.org 是已完事的，都不该出现在任务列表里
(set! agenda-exclude-files '("init.org" "archive.org"))
(set! todo-exclude-files   '("init.org" "archive.org"))
#+end_src

* Tasks 页默认过滤

#+begin_src scheme
;; 显示未完成任务，高优先级排前面
(set! todo-default-filter "/(state:TODO OR state:WAITING) group:state order:>priority")

;; App 图标角标只统计逾期且未完成的任务
(set! badge-filter "o k")
#+end_src

* 捕获模板

和桌面端一样只有两个：要做的事进 tasks.org，想法进 ideas.org。

#+begin_src scheme
(set! item-templates
  '(
    ;; 要做的事
    ("✅ 任务"
     "s TODO f tasks.org n \"[%now%]\"")

    ;; 想法 / 笔记（不指定状态：写 s NONE 的话 beorg 会把 NONE 当关键词写进标题）
    ("💭 想法"
     "f ideas.org n \"[%now%]\"")
  ))
#+end_src

* 编辑器工具栏

#+begin_src scheme
;; 编辑笔记时键盘上方的快捷按钮。
;; 左右移动用文档里的内置图标名（icon-left / icon-right）；前两个是自定义
;; 动作，文档没有对应图标，用 emoji 当标签。如果手机上显示不出来，
;; 换成 icon-list / icon-change 之类的内置名即可。
(set! editor-toolbar-items
  '(
    ("📅" (insert (date->string (current-date) "<~Y-~m-~d ~a ~H:~M>")))
    ("✅" (insert "- [ ] "))
    ("icon-left"  (backward-char))
    ("icon-right" (forward-char))
  ))
#+end_src

* 外观

#+begin_src scheme
;; 启动后默认显示哪个 tab。变量名是 ui-start-tab（不是 startup-tab，
;; 后者不存在，写了也不会报错，只是静默失效）。取值：todo / agenda / files
(set! ui-start-tab "todo")
#+end_src
