#+TITLE: beorg 配置
#+STARTUP: overview

* TODO 关键词

和桌面端保持一致：日常只用 TODO / DONE，WAITING、SOMEDAY 需要时才用。
刻意没有 NEXT。

#+begin_src scheme
(set! org-todo-action-keywords '("TODO" "WAITING" "SOMEDAY"))

(set! org-todo-done-keywords
      '("DONE" "CANCELLED"))
#+end_src

* Agenda 设置

#+begin_src scheme
;; init.org 是配置、archive.org 是已完事的，都不该出现在任务列表里
(set! agenda-exclude-files '("init.org" "archive.org"))
(set! todo-exclude-files   '("init.org" "archive.org"))
#+end_src

* Tasks 页默认过滤

#+begin_src scheme
;; 显示未完成任务，高优先级排前面
(set! todo-default-filter "/(state:TODO OR state:WAITING) group:state order:>priority")

;; App 图标角标只统计逾期且未完成的任务
(set! badge-filter "o k")
#+end_src

* 捕获模板

和桌面端一样只有两个：要做的事进 tasks.org，想法进 ideas.org。

#+begin_src scheme
(set! item-templates
  '(
    ;; 要做的事
    ("✅ 任务"
     "s TODO f tasks.org n \"[%now%]\"")

    ;; 想法 / 笔记（不指定状态：写 s NONE 的话 beorg 会把 NONE 当关键词写进标题）
    ("💭 想法"
     "f ideas.org n \"[%now%]\"")
  ))
#+end_src

* 编辑器工具栏

#+begin_src scheme
;; 编辑笔记时键盘上方的快捷按钮
(defvar editor-toolbar-items
  '(
    ("📅" (insert (date->string (current-date) "<~Y-~m-~d ~a ~H:~M>")))
    ("✅" (insert "- [ ] "))
    ("◀"  (backward-char))
    ("▶"  (forward-char))
  ))
#+end_src

* 快捷操作

#+begin_src scheme
;; 安排到今天
(define (my/schedule-today)
  (set-current-item-scheduled! (current-date)))

;; 安排到明天
(define (my/schedule-tomorrow)
  (set-current-item-scheduled!
    (date-add (current-date) 1 'day)))
#+end_src
