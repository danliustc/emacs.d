* 说明

这是一份基于 Org-mode 文学编程的 Emacs 配置。
所有代码都在这个文件里，通过 =org-babel-tangle= 生成 =config.el=。

使用方法：
1. 把这个文件放到 =~/.emacs.d/config.org=
2. 在 =~/.emacs.d/init.el= 里写：
   #+begin_src emacs-lisp :tangle no
   (org-babel-load-file (expand-file-name "config.org" user-emacs-directory))
   #+end_src
3. 先装好依赖工具（Homebrew）：
   #+begin_src sh :tangle no
   brew install ripgrep aspell pandoc
   #+end_src
4. 重启 Emacs，一切就绪。

如果你想手动 tangle：=M-x org-babel-tangle= 或 =C-c C-v t=

#+begin_src emacs-lisp
;; 用 Emacs 内置 package.el 管理第三方包，use-package 负责声明配置。
(require 'package)
(setq package-archives
      '(("gnu"    . "https://elpa.gnu.org/packages/")
        ("nongnu" . "https://elpa.nongnu.org/nongnu/")
        ("melpa"  . "https://melpa.org/packages/")))

;; 稳定优先。MELPA 是滚动快照，版本号形如 20260502.950，永远大于 GNU ELPA 的
;; 语义版本号 1.9；不设优先级的话每个包都会装成当天快照。设好之后，只有 MELPA
;; 独有的包（org-superstar、dired-sidebar）才走 MELPA。
(setq package-archive-priorities
      '(("gnu"    . 3)
        ("nongnu" . 2)
        ("melpa"  . 1)))

(package-initialize)

(unless (require 'use-package nil t)
  (error "use-package is not available. Emacs 29+ includes it; otherwise install it first."))
(setq use-package-always-ensure nil)

(defvar my/packages
  '(exec-path-from-shell
    solarized-theme
    vertico
    orderless
    marginalia
    consult
    embark
    embark-consult
    org-superstar
    markdown-mode
    dired-sidebar)
  "Third-party packages installed from the archives in `package-archives'.
which-key、use-package 是 Emacs 30 内置的，不需要列在这里。")

;; 让 package.el 知道哪些包是「要的」：package-autoremove 靠它识别孤儿包，
;; package-upgrade-all 靠它决定升级范围。这个值必须和上面那张表一致，
;; 否则 autoremove 会误判（见「收尾」一节的重新赋值）。
(setq package-selected-packages my/packages)

(defun my/install-missing-packages ()
  "Install every package this configuration needs.
装完需要重启 Emacs：各个 use-package 块的 :if 在配置加载时就已经求值了。"
  (interactive)
  (package-refresh-contents)
  (package-install-selected-packages t)
  (message "包安装完成，重启 Emacs 后生效。"))

(defun my/upgrade-packages ()
  "Upgrade all installed packages."
  (interactive)
  (package-refresh-contents)
  (package-upgrade-all)
  (message "包升级完成，重启 Emacs 后生效。"))

#+begin_src emacs-lisp
;; macOS 上 Option 键作为 Meta，保留 Command 作为 Super。
;; 这样 Option+x 是 M-x，Command 不再和 Meta 快捷键冲突。
(setq mac-command-modifier 'super
      mac-option-modifier  'meta
      mac-right-option-modifier nil)
(setq ns-command-modifier 'super
      ns-option-modifier  'meta
      ns-right-option-modifier nil)

;; 让 Emacs 能找到 Homebrew 安装的工具（ripgrep、aspell、pandoc 等）
;; macOS GUI 启动时不走 shell，所以 PATH 要手动设置
(use-package exec-path-from-shell
  :ensure nil
  :if (package-installed-p 'exec-path-from-shell)
  :config
  (when (memq window-system '(mac ns))
    (exec-path-from-shell-initialize)))

;; macOS 原生全屏（不是 Emacs 的假全屏）
(setq ns-use-native-fullscreen t)

;; 标题栏显示文件名
(setq frame-title-format '("%b — Emacs"))

;; 删除文件走废纸篓，而不是真删。
;; 刻意不设 trash-directory：Emacs 30 的 move-file-to-trash 先判
;; system-move-file-to-trash，macOS 上它有定义，所以走的是系统原生废纸篓
;; （Finder 里能「放回原处」）。设了 trash-directory 反而会绕开原生接口，
;; 变成一次普通 rename，重名还要加后缀。
(setq delete-by-moving-to-trash t)

;; 在 macOS 上禁用右键菜单弹出（避免和系统冲突）
(when (display-graphic-p)
  (context-menu-mode -1))
#+end_src

* 命令搜索 — 核心体验（Vertico + Consult + Embark + Marginalia）

- =M-x= → 搜索所有命令（模糊匹配，带描述）
- =M-X= → 只搜索当前 major-mode 的命令
- =C-h k= → 查某个快捷键绑了什么命令

* GTD — 简单好用的任务管理

** 核心思路

日常只有两个动作：

#+begin_example
记一笔  C-c c   t = 要做的事 → tasks.org
                n = 想法笔记 → ideas.org
看一眼  C-c a d 今天到期的 + 所有没排期的待办
#+end_example

三个文件：=tasks.org= 要做的、=ideas.org= 只是记的、=archive.org= 完事的。

没有 Inbox 中转、没有 NEXT 状态、没有整理仪式。想整理的时候打个标签或排个期，
不整理也照样能用——这是刻意的：以前的设计是「不整理就什么都看不见」。

** Org-mode 基础配置

#+begin_src emacs-lisp
(use-package org
  :ensure nil
  :config

  ;; --- 文件 ---
  ;; ideas.org 也放进来做保险：它里面的条目没有 TODO 关键词，不会出现在任何
  ;; 待办视图里；但万一有任务被记进去，不至于像以前那样彻底消失。
  (setq org-directory my/org-dir)
  (setq org-agenda-files
        (append (list my/org-tasks my/org-ideas)
                (when (bound-and-true-p my/org-extra-agenda-files)
                  my/org-extra-agenda-files)))

  ;; --- TODO 关键词 ---
  ;; 日常只用 TODO 和 DONE。WAITING/SOMEDAY/CANCELLED 留着，需要时才用，
  ;; 不需要时不占脑子。刻意没有 NEXT：它要求一个「整理」动作，而那个动作
  ;; 在实际使用中从来没发生过，结果是依赖它的视图长期是空的。
  (setq org-todo-keywords
        '((sequence "TODO(t)" "|" "DONE(d)")
          (sequence "WAITING(w@/!)" "SOMEDAY(s)" "|" "CANCELLED(c@)")))

  (setq org-todo-keyword-faces
        '(("TODO"      . (:foreground "#b45309" :weight bold))
          ("WAITING"   . (:foreground "#6d28d9" :weight bold))
          ("SOMEDAY"   . (:foreground "#64748b" :weight bold))
          ("DONE"      . (:foreground "#15803d"))
          ("CANCELLED" . (:foreground "#6b7280" :strike-through t))))

  ;; --- 标签 ---
  (setq org-tag-alist
        '(("personal" . ?p)
          ("work"     . ?w)
          ("home"     . ?h)
          ("errands"  . ?e)
          ("computer" . ?c)
          ("health"   . ?m)
          ("learning" . ?l)
          ("travel"   . ?v)
          ("energy_high" . ?H)
          ("energy_low"  . ?L)
          ("quick"       . ?q)))

  ;; --- 完成时自动记录时间 ---
  (setq org-log-done 'time)
  (setq org-log-into-drawer t)

  ;; --- Archive ---
  (setq org-archive-location (concat my/org-archive "::* Archived"))

  ;; --- Refile ---
  ;; tasks.org / ideas.org 是平的：每个条目都是一级标题，没有容器。
  ;; 所以在这套配置里 refile 只有一个含义：把条目整个挪到另一个文件的顶层。
  ;; 绝不能塞到某个条目下面——那会造出二级标题，而 beorg 只认一级平铺。
  ;;
  ;; 两个关键点，改之前先看清楚：
  ;;
  ;; 1. org-refile-use-outline-path 必须是 'file，不能是 t。
  ;;    org-refile.el 里只有 'file / 'title / 'buffer-name / 'full-file-path
  ;;    这几个取值会把「文件本身」推进候选列表（见 org-refile-get-targets）。
  ;;    用 t 的话文件顶层根本选不中，唯一能做的就是往某个条目下面塞，
  ;;    正好是这里要避免的事。
  ;;
  ;; 2. tasks/ideas 用一个永不匹配的 :regexp（\` 和 \' 只能在空 buffer 里
  ;;    同时成立），于是候选里只剩两个文件名，从源头杜绝误挂到别的条目下面。
  ;;    archive.org 用 :level 1，因为它确实有个 * Archived 容器要能选中。
  ;;
  ;; 顺带 org-refile-allow-creating-parent-nodes 设成 nil：目标就这几个，
  ;; 允许现造父节点只会重新打开「制造二级标题」这个口子。
  (setq org-refile-targets
        `((,my/org-tasks   :regexp . "\\`\\'")
          (,my/org-ideas   :regexp . "\\`\\'")
          (,my/org-archive :level  . 1)))
  (setq org-refile-use-outline-path 'file)
  (setq org-outline-path-complete-in-steps nil)  ; 一步完成，配合 vertico
  (setq org-refile-allow-creating-parent-nodes nil)

  ;; --- Capture ---
  ;; 只有两个模板，捕获时只需回答一个问题：这需要我做点什么吗？
  ;;   是 → t，进 tasks.org；否 → n，进 ideas.org
  ;; 不选文件、不选标签、不填时间。要整理的时候再整理，不整理也能用。
  ;; 用 (file ...) 而不是 (file+headline ...)：追加到文件末尾，一级标题。
  ;; beorg 的捕获就是这么写的（实测：它在文件末尾新建一级标题，不认容器标题），
  ;; 两端保持一致，文件里也就不需要维护任何容器结构。
  (setq org-capture-templates
        `(("t" "✅ 任务" entry
           (file ,my/org-tasks)
           "* TODO %?\n:PROPERTIES:\n:CREATED: %U\n:END:\n%i\n"
           :empty-lines 1)

          ("n" "💭 想法 / 笔记" entry
           (file ,my/org-ideas)
           "* %?\n:PROPERTIES:\n:CREATED: %U\n:END:\n%i\n"
           :empty-lines 1)))

  ;; --- 外观 ---
  (setq org-hide-emphasis-markers t    ; 隐藏 *bold* 的星号
        org-pretty-entities t          ; 显示 LaTeX 符号
        org-ellipsis " ▾"
        org-startup-indented t         ; 缩进对齐
        org-src-fontify-natively t     ; 代码块语法高亮
        org-src-tab-acts-natively t
        org-confirm-babel-evaluate t)

  )
#+end_src

** Agenda 视图 — 你的每日仪表盘

#+begin_src emacs-lisp
(with-eval-after-load 'org-agenda
  ;; 只有一个视图。C-c a d 看今天，就这一个入口。
  ;; 第一块是今天到期的（含日常杂务的 repeater）；第二块是没排期的所有待办。
  ;; SOMEDAY 刻意不显示——那正是 SOMEDAY 的意义。想看的话按 C-c a t。
  (setq org-agenda-custom-commands
        '(("d" "📅 今天"
           ((agenda ""
                    ((org-agenda-span 1)
                     (org-agenda-show-all-dates nil)
                     (org-agenda-start-on-weekday nil)))
            (todo "TODO|WAITING"
                  ((org-agenda-overriding-header "待办")
                   (org-agenda-skip-function
                    '(org-agenda-skip-entry-if 'scheduled 'deadline))))))))

  ;; Agenda 显示设置
  (setq org-agenda-window-setup 'current-window

** 初始化 org 文件目录

#+begin_src emacs-lisp
(defun my/gtd-file-title (path)
  "Return a readable Org title for PATH."
  (file-name-sans-extension (file-name-nondirectory path)))

(defun my/gtd-create-file (path &optional body)
  "Create PATH with a standard Org header and optional BODY."
  (unless (file-exists-p path)
    (with-temp-file path
      (insert (format "#+TITLE: %s\n#+STARTUP: overview\n\n%s"
                      (my/gtd-file-title path)
                      (or body ""))))))

(defun my/gtd-ensure-top-level-headings (path headings)
  "Append missing top-level HEADINGS to PATH.

PATH 可能已经被用户打开着（这个函数是 interactive 命令 `my/gtd-initialize'
的一部分，也可能碰上会话恢复出来的 buffer）。所以：
- 只 kill 自己新建的 buffer，别人开的原样留着；
- 只在自己确实改了东西的时候存盘，不替用户决定要不要保存他的未存修改。"
  (let* ((existing (find-buffer-visiting path))
         (buf (or existing (find-file-noselect path)))
         (changed nil))
    (with-current-buffer buf
      (save-excursion
        (dolist (heading headings)
          (unless (save-excursion
                    (goto-char (point-min))
                    (re-search-forward
                     (format "^\\* %s\\(?:[ \t\n]\\|$\\)" (regexp-quote heading))
                     nil t))
            (setq changed t)
            (goto-char (point-max))
            (unless (bolp) (insert "\n"))
            (insert (format "\n* %s\n" heading)))))
      (when changed
        (save-buffer)))
    (unless existing
      (kill-buffer buf))))

(defun my/gtd-initialize ()
  "创建三个 org 文件（如果不存在）。"
  (interactive)
  (make-directory my/org-dir t)
  ;; tasks.org / ideas.org 是平的：条目都是一级标题，没有容器标题。
  ;; 只有 archive.org 需要一个标题，因为 org-archive-location 指向它。
  (dolist (path (list my/org-tasks my/org-ideas my/org-archive))
    (my/gtd-create-file path))
  (my/gtd-ensure-top-level-headings my/org-archive '("Archived")))

;; 启动时自动初始化
(add-hook 'emacs-startup-hook #'my/gtd-initialize)

* 剪贴板 — 与 macOS 系统互通

macOS GUI 版 Emacs 通常能自动同步剪贴板，但终端版（=emacs -nw=）或某些情况下会断开。这节把两种情况都覆盖好。

#+begin_src emacs-lisp
;; GUI 版：直接用系统剪贴板（通常默认已开，显式写出更保险）
(setq select-enable-clipboard t
      select-enable-primary   nil)   ; macOS 没有 X11 primary selection，关掉

;; 终端版（emacs -nw）：用 pbcopy / pbpaste 桥接系统剪贴板
(unless (display-graphic-p)
  (defun my/pbcopy ()
    "把 Emacs kill-ring 最新内容复制到 macOS 系统剪贴板。"
    (interactive)
    (let ((text (current-kill 0 t)))
      (with-temp-buffer
        (insert text)
        (call-process-region (point-min) (point-max) "pbcopy"))))

  (defun my/pbpaste ()
    "把 macOS 系统剪贴板内容粘贴到 Emacs，并推入 kill-ring。"
    (interactive)
    (let ((text (shell-command-to-string "pbpaste")))
      (kill-new text)
      (insert text)))

  ;; 覆盖终端模式下的复制/粘贴行为
  (global-set-key (kbd "M-w") (lambda ()
                                (interactive)
                                (call-interactively 'kill-ring-save)
                                (my/pbcopy)))
  (global-set-key (kbd "C-w") (lambda ()
                                (interactive)
                                (call-interactively 'kill-region)
                                (my/pbcopy)))
  (global-set-key (kbd "C-y") 'my/pbpaste))

;; consult-yank-pop：M-y（Option+Y）弹出剪贴板历史，可以粘贴任意一条历史记录
;; （已在命令搜索一节绑定到 M-y，这里作为提示保留）
#+end_src

| 操作 | GUI Emacs | 终端 Emacs |
|------|-----------|------------|
| 复制到系统剪贴板 | =M-w= | =M-w=（自动调 pbcopy）|
| 从系统剪贴板粘贴 | =C-y= | =C-y=（自动调 pbpaste）|
